# successor_trial.sh - CDR-012 fresh successor of combined trial #39 (Primary Integration Writer).
#
# Adapted from the hardened CDR-011 harness next_trial.sh (sha256 2e2d45f6..., the harness that
# produced trial #39). Builds, in a throwaway detached worktree of a PRIVATE clone,
#     T = (#39 tip)  --no-ff merge  (#31 head)  --no-ff merge  (#35 head)
# then runs the checks listed by --list-checks, writes summary.json and deletes the private clone.
#
# Isolation: the harness never creates a worktree in, fetches into, or writes to the source
# repository (--src-repo, default /home/user/Investment-System1). It clones it into
# WORK/repo (git clone --no-checkout --shared; a shallow source is copied instead), points that
# clone's origin at the real remote, fetches (and unshallows) there, and builds every worktree and
# merge inside the clone. The whole WORK directory is deleted at exit unless --keep.
#
# Merge order (documented choice): #31 first, then #35.
#   * Codex stacks #35 on #31. The harness checks whether the #31 head is an ancestor of the #35
#     head. STACKED (ancestor): same convention as #39. INDEPENDENT (#35 has not absorbed the new
#     #31): the harness still merges #31 and then #35, reports the topology prominently and checks
#     the result against an independent merge-tree replay instead of a direct #39+#35 merge.
#   * An input already contained in the current trial HEAD is not merged again (git would say
#     "Already up to date"); it is recorded as NOOP_ALREADY_CONTAINED.
#   * No-reuse rule (CDR-011/CDR-012): if the trial tree equals #39's tree the run is labelled
#     NO_NEW_CONTENT, it is never called a successor trial, and the remaining checks are recorded
#     NOT_RUN unless --force-checks is given.
#
# READ-ONLY on GitHub and on every remote ref: only `git fetch` / `git ls-remote` run against the
# remote. It never pushes, comments, dispatches or re-runs anything. It never reads CAL_VERIFY or
# Holdout data and never fetches provider data. Track C C8 paths and codex/* branches are merged
# for verification only; the harness never edits any repository file.
#
# Usage:
#   successor_trial.sh --label LABEL --verified-v2-commit SHA [options] SHA39 SHA31 SHA35
#     LABEL   DRY_RUN_* (harness validation) or e.g. CDR012_SUCCESSOR_TRIAL (the real trial).
#             A non-DRY_RUN label is refused while #31 is still at the pre-CDR-012 head e0b6d80,
#             while VERIFIED_V2_COMMIT is e0b6d80, or when SHA39 is not the verified #39 tip 0d31e06.
#   required:
#     --verified-v2-commit SHA   commit whose evl/superiority_v2.py (and reviewed allowlist files)
#                                were independently verified; no default. Must be an ancestor of
#                                (or equal to) the #31 or #35 head.
#   options:
#     --reviewed-allowlist FILE  reviewed Track C additive allowlist, one repo path per line ('#'
#                                comments). Default: the 11 entries verified for #39. The trial's
#                                declared CI allowlist must equal it; any addition outside it FAILs.
#     --reviewed-c8-evidence FILE  reviewed C8-scope additions that are not in the allowlist (reports
#                                / tools). Default: the 2 approval reports present in #39.
#     --min-tests N    full-suite lower bound (default 1441 = #39 count; tests are only added)
#     --force-checks   run every check even when the trial tree equals #39's (NO_NEW_CONTENT)
#     --oracle PATH    independent CDR-012 oracle; NOT_RUN when the file is absent. Default:
#                      .../wf11/prep-cdr012-oracle/compare_cdr012.py, else .../prep-cdr012-oracle/base/compare_cdr012.py
#                      (the path and sha256 actually used are recorded). Run with PYTHONPATH=<oracle dir>:<trial src>,
#                      PYTHONDONTWRITEBYTECODE=1, cwd = the run directory. PASS needs exit 0 and, if it writes JSON,
#                      status PASS or verdict EXACT_MATCH_ALL_GRADED.
#     --oracle-python PY   interpreter for the oracle (default: the harness venv python)
#     --oracle-args STR    oracle arguments; placeholders {src} {wt} {out} {oracle_dir}
#                          (default: '--adapter v2kernel --out {out}', the compare_cdr012.py CLI)
#     --port N         private HTTP port for the two browser tests that hard-code 8765 (default 8792)
#     --only a,b,...   run only these checks (setup/build steps always run)
#     --skip a,b,...   skip these checks (recorded NOT_RUN)
#     --keep           keep the private clone, worktrees and work directory
#     --venv DIR       Python 3.11 venv with numpy==2.3.5 + pytest==9.1.1 (created if absent)
#     --base DIR       harness base directory (default: directory of this script)
#     --src-repo DIR   read-only clone source (default /home/user/Investment-System1)
#     --base-branch B / --br31 B / --br35 B   branch-name overrides, DRY_RUN_* labels only (self-tests)

def load_list(env, default):
    """Reviewed path list from a file named by env (one path per line, '#' comments) or the default."""
    f = os.environ.get(env)
    if not f:
        return default, "default"
    paths = set()
    for raw in Path(f).read_text().splitlines():
        line = raw.split("#", 1)[0].strip()
        if not line:
            continue
        if not re.fullmatch(r"implementation/[A-Za-z0-9_./-]+", line) or ".." in line:
            raise SystemExit(f"bad path in {f}: {raw!r}")
        paths.add(line)
    if not paths:
        raise SystemExit(f"{f} lists no paths")
    return frozenset(paths), f"{f} (sha256 {hashlib.sha256(Path(f).read_bytes()).hexdigest()})"


def git(repo, *args, binary=False):
    p = subprocess.run(["git", "-C", str(repo), *args], capture_output=True)
    if p.returncode != 0:
        raise RuntimeError(f"git {' '.join(args)} rc={p.returncode}: {p.stderr.decode(errors='replace')}")
    return p.stdout if binary else p.stdout.decode()


def ls_tree(repo, ref):
    res = {}
    for rec in git(repo, "ls-tree", "-r", "--full-tree", "-z", ref).split("\0"):
        if rec:
            meta, path = rec.split("\t", 1)
            res[path] = meta.split()[2]
    return res


def blob(repo, ref, path):
    return git(repo, "cat-file", "blob", f"{ref}:{path}", binary=True)


def is_ancestor(repo, a, b):
    return subprocess.run(["git", "-C", str(repo), "merge-base", "--is-ancestor", a, b]).returncode == 0

def parse_allowlist(text):
    m = re.search(r"allowed_additions\s*=\s*\{(.*?)\}", text, re.S)
    if m is None:
        return None
    return {a or b for a, b in re.findall(r"'([^']+)'|\"([^\"]+)\"", m.group(1))}


def rc_of(status):
    return 0 if status in ("PASS", "INCOMPLETE", "REVIEW_REQUIRED") else 1


def read_merges(path):
    rows = []
    for line in Path(path).read_text().splitlines():
        pr, sha, rc, head, action, parent = (line.split("\t") + [""] * 6)[:6]
        rows.append({"pr": pr, "sha": sha, "rc": rc, "head": head, "action": action, "parent": parent})
    return rows


# ------------------------------------------------------------------ junit
def junit_summary(path):
    root = ET.parse(path).getroot()
    tot = collections.Counter()
    per_mod = collections.defaultdict(collections.Counter)
    failing = []
    for c in root.iter("testcase"):
        cls = c.get("classname", "")
        parts = cls.split(".")
        mod = parts[1] if parts[0] == "tests" and len(parts) > 1 else parts[0]
        if c.find("failure") is not None:
            k = "failed"
        elif c.find("error") is not None:
            k = "error"
        elif c.find("skipped") is not None:
            k = "skipped"
        else:
            k = "passed"
        tot[k] += 1
        tot["total"] += 1
        per_mod[mod][k] += 1
        per_mod[mod]["total"] += 1
        if k in ("failed", "error"):
            failing.append(f"{cls}::{c.get('name')}")
    groups = {
        "track_c_c8_v1": lambda m: m.startswith("test_evl_c8"),
        "track_c_gsup_additive": lambda m: m.startswith("test_evl_gsup"),
        "track_c_all_test_evl": lambda m: m.startswith("test_evl_"),
        "producer_source_compat": lambda m: m == "test_producer_source_compat",
        "producer_infra_compat": lambda m: m.endswith("producer_infra_compat") or m == "test_c28_integration_compat",
        "web_research_guard": lambda m: m == "test_web_research_guard",
        "web_state_presentation": lambda m: m == "test_web_state_presentation",
        "web_mvp_and_search": lambda m: m in ("test_web_mvp", "test_global_language_search", "test_entity_metadata_coverage"),
    }
    g = {}
    for name, pred in groups.items():
        c = collections.Counter()
        for m, cnt in per_mod.items():
            if pred(m):
                c.update(cnt)
        g[name] = dict(c)
    return {"totals": dict(tot), "groups": g, "failing": failing, "modules": len(per_mod)}


def cmd_junit(args):
    path, out = args
    s = junit_summary(path)
    t = s["totals"]
    min_tests = int(os.environ["MIN_TESTS"])  # lower bound; tests are only ever added
    status = "PASS" if (t.get("total", 0) >= min_tests and t.get("failed", 0) == 0 and t.get("error", 0) == 0
                        and t.get("skipped", 0) == 0) else "FAIL"
    emit(out, {"status": status, "junit": path, "counts": t, "min_tests": min_tests, "skipped_must_be_zero": True, **s})
    return 0 if status == "PASS" else 1

def cmd_extract_step(args):
    import yaml
    wf, step_name, out = args
    doc = yaml.safe_load(Path(wf).read_text())
    for job_id, job in (doc.get("jobs") or {}).items():
        for st in job.get("steps") or []:
            if st.get("name") == step_name:
                Path(out).write_text(st["run"])
                print(json.dumps({"workflow": wf, "job": job_id, "step": step_name, "bytes": len(st["run"]),

# ------------------------------------------------------------------ tree identity (merge replay)
def merge_tree(wt, a, b):
    p = subprocess.run(["git", "-C", str(wt), "-c", "gc.auto=0", "merge-tree", "--write-tree", a, b], capture_output=True)
    return p.returncode, p.stdout.decode().split("\n", 1)[0].strip()


def changed_paths(wt, a, b):
    return set(git(wt, "diff", "--no-renames", "--name-only", a, b).split("\n")) - {""}


def cmd_tree_identity(args):
    wt, trial, base, sha31, sha35, merges, topology, out = args
    items, flags = [], []
    t = git(wt, "rev-parse", f"{trial}^{{tree}}").strip()
    rows = read_merges(merges)

             (("base_39", base), ("pr31", sha31), ("pr35", sha35), ("trial", trial))}
    # every real merge changes only paths that its input changed since a merge base with the parent
    for r in rows:
        if r["action"] != "MERGED":
            items.append({"item": f"merge PR #{r['pr']}: {r['action']}", "status": "PASS", "head": r["head"]})
            continue

        delta = changed_paths(wt, r["parent"], r["head"])

        items.append({"item": f"merge PR #{r['pr']}: changes only paths the input changed since its merge base(s)",
                      "status": "PASS" if allowed is not None and not extra else "FAIL",
                      "merge_bases": mbs, "criss_cross": len(mbs) > 1, "delta_paths": len(delta),
                      "input_delta_paths": len(allowed or ()), "outside_input_delta": extra[:50]})
    if topology == "STACKED":
        rc, direct = merge_tree(wt, base, sha35)
        items.append({"item": "STACKED: trial tree equals a direct merge-tree of #35 into #39", "status":
                      "PASS" if rc == 0 and direct == t else "FAIL", "trial_tree": t, "direct_tree": direct, "rc": rc})
    else:
        flags.append({"flag": "PR35_DID_NOT_ABSORB_PR31", "detail": "#31 head is not an ancestor of #35 head; merged #31 then #35"})
        rc1, t1 = merge_tree(wt, base, sha31)

# ------------------------------------------------------------------ CDR-012 F1 static scan (AST; read-only)
POW_CALLS = {"pow", "power", "float_power", "square"}


def pow_findings(source):
    tree = ast.parse(source)
    found = []
    for node in ast.walk(tree):
        kind = None
        if isinstance(node, ast.BinOp) and isinstance(node.op, ast.Pow):
            r = node.right
            kind = "SQUARE_BY_POW" if isinstance(r, ast.Constant) and r.value in (2, 2.0) and not isinstance(r.value, bool) else "OTHER_POW"
        elif isinstance(node, ast.AugAssign) and isinstance(node.op, ast.Pow):
            kind = "AUGASSIGN_POW"
        elif isinstance(node, ast.Call):
            f = node.func
            name = f.id if isinstance(f, ast.Name) else f.attr if isinstance(f, ast.Attribute) else None
            if name in POW_CALLS:
                kind = "POW_CALL"
        if kind:
            found.append({"kind": kind, "line": node.lineno, "col": node.col_offset,
                          "code": ast.get_source_segment(source, node)})
    selfmul = sum(1 for n in ast.walk(tree) if isinstance(n, ast.BinOp) and isinstance(n.op, ast.Mult)
                  and ast.dump(n.left) == ast.dump(n.right))
    return sorted(found, key=lambda d: (d["line"], d["col"])), selfmul

def cmd_escalation(args):
    wt, base, trial, scope, prefix = args
    raw = subprocess.run(["git", "-C", wt, "diff", "-U0", "--no-color", "--no-ext-diff", "--no-renames", base, trial],
                         capture_output=True, check=True).stdout.decode("utf-8", errors="replace")
    comp = [(n, re.compile(p)) for n, p in PATTERNS]
    hits, path, ln = [], None, 0
    for line in raw.split("\n"):
        if line.startswith("+++ "):
            path = line[6:] if line.startswith("+++ b/") else None
            continue
        if line.startswith("@@"):
            m = re.match(r"@@ -\S+ \+(\d+)", line)
            ln = int(m.group(1)) if m else 0
            continue
        if line.startswith("+") and path is not None:
            text = line[1:]
            for name, rx in comp:
                for k, m in enumerate(rx.finditer(text)):
                    if k >= 20:
                        break
                    ctx = text[max(0, m.start() - 120): m.end() + 120]
                    hits.append({"path": path, "line": ln, "category": category(path), "pattern": name,
                                 "likely_negated": bool(NEG.search(ctx)), "context": ctx})
            ln += 1
    counts = collections.Counter((h["category"], h["pattern"]) for h in hits)
    code = [h for h in hits if h["category"] in ("src", "tools", "ci")]
    code_unneg = [h for h in code if not h["likely_negated"]]
    unneg = [h for h in hits if not h["likely_negated"]]
    by_cat = collections.Counter(h["category"] for h in hits)
    with open(prefix + ".tsv", "w") as f:
        f.write("scope\tcategory\tpattern\tlikely_negated\tpath\tline\tcontext\n")
        for h in hits:
            f.write("\t".join([scope, h["category"], h["pattern"], str(h["likely_negated"]), h["path"], str(h["line"]),
                               h["context"].replace("\t", " ").replace("\r", " ")]) + "\n")
    status = "NO_HITS" if not hits else "HITS_LISTED_FOR_REVIEW"
    emit(prefix + ".json", {"status": status, "scope": scope, "base": base, "trial": trial,
                            "counts": {"total": len(hits), "not_negated": len(unneg), "by_category": dict(by_cat),
                                       "code_ci_hits": len(code), "code_ci_hits_not_negated": len(code_unneg)},
                            "by_category_pattern": {f"{c}|{p}": n for (c, p), n in sorted(counts.items())},
                            "not_negated_hits": unneg, "code_ci_hits": code, "hits_tsv": prefix + ".tsv"})
    return 0

def glob_re(p):
    out, i = "", 0
    while i < len(p):
        if p.startswith("**", i):
            out += ".*"
            i += 2
        elif p[i] == "*":
            out += "[^/]*"
            i += 1
        elif p[i] == "?":
            out += "[^/]"
            i += 1
        else:
            out += re.escape(p[i])
            i += 1
    return re.compile("^" + out + "$")


def filter_match(patterns, files):
    rules = [(p.startswith("!"), glob_re(p[1:] if p.startswith("!") else p)) for p in patterns]
    hit = []
    for f in files:
        ok = False
        for neg, rx in rules:
            if rx.match(f):
                ok = not neg
        if ok:
            hit.append(f)
    return hit

def cmd_pathfilters(args):

    wt, trial, out, *bases = args
    wfs = {}
    for path in git(wt, "ls-tree", "--name-only", trial, ".github/workflows/").split():
        doc = yaml.safe_load(blob(wt, trial, path))
        on = doc.get("on", doc.get(True))
        if isinstance(on, str):
            on = {on: None}
        elif isinstance(on, list):
            on = {k: None for k in on}
        wfs[path.rsplit("/", 1)[1]] = on or {}
    result = {}
    for spec in bases:
        name, sha = spec.split("=", 1)
        mb = git(wt, "merge-base", sha, trial).strip()

                m = filter_match(pr["paths"], files)

def cmd_fingerprints(args):

        items.append({"item": f"{name}: tool output sha256 and values", "status": "PASS" if sha == EXPECTED_FP_SHA and not diff else "FAIL",
                      "output_sha256": sha, "expected_sha256": EXPECTED_FP_SHA, "technical": got.get("technical"),
                      "macro": got.get("macro"), "values": got, "mismatches": diff})
    emit(out, {"status": overall(items), "items": items})

# ---------------------------------------------------------------- step runner
selected() {
  local n=$1
  if [ -n "$ONLY" ] && ! [[ ",$ONLY," == *",$n,"* ]]; then return 1; fi
  if [ -n "$SKIP" ] && [[ ",$SKIP," == *",$n,"* ]]; then return 1; fi
  return 0
}

record() { # name status rc seconds log note
  STATUS[$1]=$2
  printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$1" "$2" "$3" "$4" "$5" "${6//$'\t'/ }" >> "$STEPS"
}

# run_step NAME FATAL DEPS(comma list or -) FUNC [ARGS...]
run_step() {
  local name=$1 fatal=$2 deps=$3; shift 3
  local always=0
  case "$name" in env_setup|private_clone_fetch|verify_inputs|verify_baselines|build_trial) always=1 ;; esac
  if [ "$always" = 0 ] && ! selected "$name"; then record "$name" NOT_RUN - 0 - "not selected (--only/--skip)"; return 0; fi
  if [ "$FATAL" = 1 ]; then record "$name" NOT_RUN - 0 - "earlier fatal step failed"; return 0; fi
  if [ "$NNC" = 1 ] && [ "$FORCE" = 0 ] && [ "$always" = 0 ] && [[ "$STRUCTURAL_CHECKS" != *",$name,"* ]]; then
    record "$name" NOT_RUN - 0 - "NO_NEW_CONTENT: trial tree equals #39 tree; not re-run (pass --force-checks to run)"; return 0
  fi
  local d
  for d in ${deps//,/ }; do
    [ "$d" = "-" ] && continue
    if [ "${STATUS[$d]:-NOT_RUN}" != PASS ]; then record "$name" NOT_RUN - 0 - "dependency $d is ${STATUS[$d]:-NOT_RUN}"; return 0; fi
  done
  STEPNO=$((STEPNO + 1))
  local logf; logf="$OUT/logs/$(printf '%02d' "$STEPNO")_${name}.log"
  log "BEGIN $name"
  local t0 t1 rc
  t0=$(date +%s.%N)
  set +e
  ( set -euo pipefail; "$@" ) > "$logf" 2>&1
  rc=$?
  set -e
  t1=$(date +%s.%N)
  local secs; secs=$("$SYSPY" -c "print(round($t1-$t0,1))")
  local status=FAIL
  local rj="$OUT/results/${name}.json"
  if [ "$rc" -eq 0 ]; then
    status=PASS
    if [ -f "$rj" ]; then
      local s; s=$("$SYSPY" -c 'import json,sys; print(json.load(open(sys.argv[1])).get("status","PASS"))' "$rj")
      # a result file may only refine a PASS (e.g. REVIEW_REQUIRED, NO_NEW_CONTENT); it can never turn rc!=0 into PASS
      [ -n "$s" ] && status=$s
    fi
  fi
  local note=""
  [ -f "$OUT/results/${name}.note" ] && note=$(head -n1 "$OUT/results/${name}.note")
  record "$name" "$status" "$rc" "$secs" "$logf" "$note"
  log "END   $name rc=$rc status=$status ${secs}s log=$logf ${note:+($note)}"
  if [ "$rc" -ne 0 ]; then
    log "----- last 25 lines of $logf (display only; rc already recorded) -----"
    tail -n 25 "$logf" | sed 's/^/    | /' >&2 || true
    if [ "$fatal" = 1 ]; then FATAL=1; log "FATAL: $name failed; remaining steps will be NOT_RUN"; fi
  fi
  return 0
}

step_private_clone_fetch() {
  local url a b
  url=$(git -C "$SRC_REPO" remote get-url "$REMOTE")
  [[ "$url" =~ $EXPECTED_ORIGIN_RE ]] || { echo "unexpected origin URL in source repo: $url"; return 1; }

  git -C "$PRIV" remote set-url "$REMOTE" "$url"
  git -C "$PRIV" config gc.auto 0
  git -C "$PRIV" config maintenance.auto false

  b=$(readlink -f "$(git -C "$SRC_REPO" rev-parse --path-format=absolute --git-common-dir)")
  [ "$a" != "$b" ] || { echo "ISOLATION: private clone shares the source git dir"; return 1; }

remote_head() { git -C "$PRIV" ls-remote "$REMOTE" "refs/heads/$1" | cut -f1; }
input_sha() { cut -f3 "$OUT/results/inputs.tsv" | sed -n "${1}p"; }

step_verify_inputs() {
  local ok=0 pr br arg full remote tracking
  : > "$OUT/results/inputs.tsv"
  for spec in "39|$BR39|$ARG39" "31|$BR31|$ARG31" "35|$BR35|$ARG35"; do
    IFS='|' read -r pr br arg <<< "$spec"
    full=$(git -C "$PRIV" rev-parse --verify --quiet "${arg}^{commit}") || { echo "PR#$pr: $arg does not resolve"; ok=1; continue; }
    remote=$(remote_head "$br")
    tracking=$(git -C "$PRIV" rev-parse "refs/remotes/$REMOTE/$br")
    printf '%s\t%s\t%s\t%s\t%s\n' "$pr" "$br" "$full" "$remote" "$tracking" | tee -a "$OUT/results/inputs.tsv"
    if [ "$full" != "$remote" ] || [ "$full" != "$tracking" ]; then echo "PR#$pr head mismatch: arg=$full ls-remote=$remote tracking=$tracking"; ok=1; fi
  done
  [ $ok -eq 0 ] || return 1
  local s39 s31 s35 v2
  s39=$(input_sha 1); s31=$(input_sha 2); s35=$(input_sha 3)
  v2=$(git -C "$PRIV" rev-parse --verify --quiet "${VERIFIED_V2_ARG}^{commit}") || { echo "VERIFIED_V2_COMMIT $VERIFIED_V2_ARG does not resolve"; return 1; }
  echo "$v2" > "$OUT/results/verified_v2_commit.txt"
  echo "VERIFIED_V2_COMMIT $v2"
  if git -C "$PRIV" merge-base --is-ancestor "$v2" "$s31" || git -C "$PRIV" merge-base --is-ancestor "$v2" "$s35"; then
    echo "OK: VERIFIED_V2_COMMIT is an ancestor of (or equal to) the #31 or #35 head"
  else
    echo "FAIL: VERIFIED_V2_COMMIT $v2 is not in the history of #31 $s31 or #35 $s35"; return 1
  fi
  if [[ "$LABEL" != DRY_RUN* ]]; then
    [ "$s31" != "$PREV31" ] || { echo "REFUSED: label $LABEL is not a dry run but #31 is still at the pre-CDR-012 head $PREV31"; return 1; }
    [ "$v2" != "$PREV31" ] || { echo "REFUSED: label $LABEL is not a dry run but VERIFIED_V2_COMMIT is the pre-CDR-012 head $PREV31"; return 1; }
    [ "$s39" = "$PR39_TIP" ] || { echo "REFUSED: label $LABEL is not a dry run but base $s39 is not the CI-verified #39 tip $PR39_TIP"; return 1; }
    # PIW 2026-10-04: a real trial must run on exactly the independently verified #31 head (GIE-011 critic gap)
    [ "$s31" = "$v2" ] || { echo "REFUSED: label $LABEL is not a dry run but #31 head $s31 is not the verified v2 commit $v2"; return 1; }
  fi
  if git -C "$PRIV" merge-base --is-ancestor "$s31" "$s35"; then
    TOPOLOGY=STACKED
    echo "OK: #31 is an ancestor of #35 (STACKED: #35 has absorbed #31)"
  else
    TOPOLOGY=INDEPENDENT
    echo "WARNING: #31 $s31 is NOT an ancestor of #35 $s35 (INDEPENDENT: #35 has not absorbed the new #31); the harness merges #31 then #35 and reports it"
  fi
  echo "$TOPOLOGY" > "$OUT/results/topology.txt"
  for c in "$s39" "$s31" "$s35"; do
    git -C "$PRIV" merge-base --is-ancestor "$CANON" "$c" || { echo "FAIL: canonical $CANON not ancestor of $c"; return 1; }
    git -C "$PRIV" merge-base --is-ancestor "$OWNER_TC" "$c" || { echo "FAIL: Track C owner $OWNER_TC not ancestor of $c"; return 1; }
  done
  echo "OK: canonical b8e39a2 and Track C b9e01a9 are ancestors of all three heads"
  local anc
  for spec in "#31_contains_prev31_e0b6d80|$PREV31|$s31" "#35_contains_prev35_8318b78|$PREV35|$s35" \
              "#39_contains_#31|$s31|$s39" "#39_contains_#35|$s35|$s39"; do
    IFS='|' read -r what a c <<< "$spec"
    if git -C "$PRIV" merge-base --is-ancestor "$a" "$c"; then anc=yes; else anc=no; fi
    echo "$what: $anc"
  done
  echo "merge-base(#39,#31)=$(git -C "$PRIV" merge-base --all "$s39" "$s31" | tr '\n' ' ')"
  echo "merge-base(#39,#35)=$(git -C "$PRIV" merge-base --all "$s39" "$s35" | tr '\n' ' ')"
  echo "#31 commits not in #39: $(git -C "$PRIV" rev-list --count "$s39..$s31"); #35 commits not in #39: $(git -C "$PRIV" rev-list --count "$s39..$s35")"
  echo "TOPOLOGY=$TOPOLOGY ($(git -C "$PRIV" rev-list --count "$s31..$s35") #35-only commits beyond #31)" > "$OUT/results/verify_inputs.note"
}

step_verify_baselines() {
  local ok=0 got what br want
  : > "$OUT/results/baselines.tsv"
  for spec in "canonical|$BR_CANON|$CANON" "trackc_owner|$BR_OWNER_TC|$OWNER_TC" "pr30_base|$BR30|$PR30"; do
    IFS='|' read -r what br want <<< "$spec"
    got=$(remote_head "$br")
    printf '%s\t%s\t%s\t%s\n' "$what" "$br" "$want" "$got" | tee -a "$OUT/results/baselines.tsv"
    [ "$got" = "$want" ] || { echo "BASELINE MOVED: $what $br expected $want got $got"; ok=1; }
  done
  git -C "$PRIV" cat-file -e "$MACRO_PIN^{commit}" && echo "macro pin $MACRO_PIN present"
  local s39; s39=$(input_sha 1)
  if [ "$s39" = "$PR39_TIP" ]; then echo "base is the CI-verified #39 tip $PR39_TIP"; else echo "NOTE: base $s39 is not the CI-verified #39 tip $PR39_TIP"; fi
  for c in "$PREV31" "$PREV35"; do
    if git -C "$PRIV" merge-base --is-ancestor "$c" "$s39"; then echo "base contains #39 constituent $c"; else echo "NOTE: base does not contain #39 constituent $c"; fi
  done
  return $ok
}

step_build_trial() {
  local s39 s31 s35
  s39=$(input_sha 1); s31=$(input_sha 2); s35=$(input_sha 3)
  git -C "$PRIV" worktree add --detach "$WT" "$s39"
  cd "$WT"
  export GIT_AUTHOR_DATE="$MERGE_DATE" GIT_COMMITTER_DATE="$MERGE_DATE"
  local G=(git -c gc.auto=0 -c maintenance.auto=false -c commit.gpgsign=false -c user.name="CDR-012 successor trial harness" -c user.email="trial-harness@invalid")
  local pr sha br out rc parent
  : > "$OUT/results/merges.tsv"
  for spec in "31|$s31|$BR31" "35|$s35|$BR35"; do
    IFS='|' read -r pr sha br <<< "$spec"
    parent=$(git rev-parse HEAD)
    if git merge-base --is-ancestor "$sha" HEAD; then
      echo "PR #$pr $sha is already contained in trial HEAD $parent: NOOP_ALREADY_CONTAINED (no merge commit)"
      printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$pr" "$sha" 0 "$parent" NOOP_ALREADY_CONTAINED "$parent" | tee -a "$OUT/results/merges.tsv"
      continue
    fi
    set +e
    out=$("${G[@]}" merge --no-ff --no-edit -m "[$LABEL] Merge PR #$pr $br @ $sha into CDR-012 successor trial of #39" "$sha" 2>&1)
    rc=$?
    set -e
    printf '%s\n' "$out"
    printf '%s\n' "$out" | grep -E '^Auto-merging ' > "$OUT/results/automerged_pr$pr.txt" || true
    printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$pr" "$sha" "$rc" "$(git rev-parse HEAD)" MERGED "$parent" | tee -a "$OUT/results/merges.tsv"
    if [ "$rc" -ne 0 ]; then
      git diff --name-only --diff-filter=U | tee "$OUT/results/conflicts_pr$pr.txt"
      "${G[@]}" merge --abort || true
      echo "MERGE CONFLICT merging PR #$pr"; return 1
    fi
    [ "$(git rev-parse HEAD^2)" = "$sha" ] && [ "$(git rev-parse HEAD^1)" = "$parent" ] || { echo "merge commit parents unexpected"; return 1; }
  done
  echo "trial commit $(git rev-parse HEAD) tree $(git rev-parse 'HEAD^{tree}')"
  echo "merges: $(cut -f5 "$OUT/results/merges.tsv" | tr '\n' ' ')" > "$OUT/results/build_trial.note"
}

step_intree_writes() {

}

step_remote_recheck() {
  local moved=0 pr br want got what
  while IFS=$'\t' read -r pr br want _ _; do
    got=$(remote_head "$br"); echo "PR#$pr $br start=$want now=$got"
    [ "$got" = "$want" ] || { echo "HEAD MOVED: $br"; moved=1; }
  done < "$OUT/results/inputs.tsv"
  if [ -f "$OUT/results/baselines.tsv" ]; then
    while IFS=$'\t' read -r what br want _; do
      got=$(remote_head "$br"); echo "$what $br expected=$want now=$got"
      [ "$got" = "$want" ] || { echo "BASELINE MOVED: $br"; moved=1; }
    done < "$OUT/results/baselines.tsv"
  else
    echo "baselines.tsv absent (verify_baselines did not run): baselines not re-checked"
    echo "baselines not re-checked (verify_baselines did not run)" > "$OUT/results/remote_recheck.note"
  fi
  return $moved
}
