# next_trial.sh - CDR-011 combined integration trial harness (Primary Integration Writer).
#
# Builds, in a throwaway detached worktree,
#     T = (#38 tip)  --no-ff merge  (#31 tip)  --no-ff merge  (#35 tip)
# whose new objects go to a temporary object directory (the main repository's
# objects are used read-only as alternates). It then runs the checks listed by
# --list-checks, writes summary.json, and deletes the worktrees and the
# temporary object directory.
#
# Merge order (documented choice): #31 first, then #35.
#   * #35 (codex/integration-hardening-2026-10-03) is stacked on #31; the harness
#     asserts that the #31 head is an ancestor of the #35 head (fatal otherwise).
#   * Merging #31 and then #35 gives one --no-ff merge commit per constituent PR,
#     each with that PR's exact head as second parent (same convention as #38),
#     and attributes any conflict to the PR that introduced it.
#   * Because #31 is an ancestor of #35, the second merge applies exactly #35's
#     own delta (#31..#35). The harness cross-checks that the final tree equals
#     a single direct merge of #35 into #38 (git merge-tree --write-tree), and that
#     the second merge's file delta equals #31..#35.
#
# READ-ONLY on GitHub and on every remote ref: it only runs `git fetch` and
# `git ls-remote`. It never pushes, comments, dispatches or re-runs anything.
# It never reads CAL_VERIFY or Holdout data and never fetches provider data.
# Track C C8 paths and codex/* branches are merged for verification only; the
# harness never edits any repository file.
#
# Usage:
#   next_trial.sh --label LABEL [options] SHA38 SHA31 SHA35
#     LABEL   DRY_RUN_PRE_V2 (harness validation) or CDR011_TRIAL (the real trial).
#             A non-DRY_RUN label is refused while #31 is still at the pre-v2 head.

def git(repo, *args, binary=False):
    p = subprocess.run(["git", "-C", str(repo), *args], capture_output=True)
    if p.returncode != 0:
        raise RuntimeError(f"git {' '.join(args)} rc={p.returncode}: {p.stderr.decode(errors='replace')}")
    return p.stdout if binary else p.stdout.decode()


def ls_tree(repo, ref):
    res = {}
    for rec in git(repo, "ls-tree", "-r", "--full-tree", "-z", ref).split("\0"):
        if rec:
            meta, path = rec.split("\t", 1)
            res[path] = meta.split()[2]
    return res


def blob(repo, ref, path):
    return git(repo, "cat-file", "blob", f"{ref}:{path}", binary=True)

def parse_allowlist(text):
    m = re.search(r"allowed_additions\s*=\s*\{(.*?)\}", text, re.S)

    return {a or b for a, b in re.findall(r"'([^']+)'|\"([^\"]+)\"", m.group(1))}

# ------------------------------------------------------------------ junit
def junit_summary(path):
    root = ET.parse(path).getroot()
    tot = collections.Counter()
    per_mod = collections.defaultdict(collections.Counter)
    failing = []
    for c in root.iter("testcase"):
        cls = c.get("classname", "")
        parts = cls.split(".")
        mod = parts[1] if parts[0] == "tests" and len(parts) > 1 else parts[0]
        if c.find("failure") is not None:
            k = "failed"
        elif c.find("error") is not None:
            k = "error"
        elif c.find("skipped") is not None:
            k = "skipped"
        else:
            k = "passed"
        tot[k] += 1
        tot["total"] += 1
        per_mod[mod][k] += 1
        per_mod[mod]["total"] += 1
        if k in ("failed", "error"):
            failing.append(f"{cls}::{c.get('name')}")
    groups = {
        "track_c_c8_v1": lambda m: m.startswith("test_evl_c8"),
        "track_c_gsup_additive": lambda m: m.startswith("test_evl_gsup"),
        "track_c_all_test_evl": lambda m: m.startswith("test_evl_"),
        "producer_source_compat": lambda m: m == "test_producer_source_compat",
        "producer_infra_compat": lambda m: m.endswith("producer_infra_compat") or m == "test_c28_integration_compat",
        "web_research_guard": lambda m: m == "test_web_research_guard",
        "web_state_presentation": lambda m: m == "test_web_state_presentation",
        "web_mvp_and_search": lambda m: m in ("test_web_mvp", "test_global_language_search", "test_entity_metadata_coverage"),
    }
    g = {}
    for name, pred in groups.items():
        c = collections.Counter()
        for m, cnt in per_mod.items():
            if pred(m):
                c.update(cnt)
        g[name] = dict(c)
    return {"totals": dict(tot), "groups": g, "failing": failing, "modules": len(per_mod)}


def cmd_junit(args):
    path, out = args
    s = junit_summary(path)
    t = s["totals"]
    min_tests = int(os.environ.get("MIN_TESTS", "1"))  # HARDENED: lower bound; tests are only ever added
    status = "PASS" if (t.get("total", 0) >= min_tests and t.get("failed", 0) == 0 and t.get("error", 0) == 0
                        and t.get("skipped", 0) == 0) else "FAIL"
    emit(out, {"status": status, "junit": path, "counts": t, "min_tests": min_tests, "skipped_must_be_zero": True, **s})
    return 0 if status == "PASS" else 1


# ------------------------------------------------------------------ workflow step extraction
def cmd_extract_step(args):
    import yaml
    wf, step_name, out = args
    doc = yaml.safe_load(Path(wf).read_text())
    for job_id, job in (doc.get("jobs") or {}).items():
        for st in job.get("steps") or []:
            if st.get("name") == step_name:
                Path(out).write_text(st["run"])
                print(json.dumps({"workflow": wf, "job": job_id, "step": step_name, "bytes": len(st["run"]),
                                  "sha256": hashlib.sha256(st["run"].encode()).hexdigest()}))
                return 0
    print(f"step not found: {step_name!r} in {wf}", file=sys.stderr)
    return 1

    return ("ci" if path.startswith(".github/") else "src" if path.startswith("implementation/src/") else

def cmd_escalation(args):
    wt, base, trial, scope, prefix = args
    raw = subprocess.run(["git", "-C", wt, "diff", "-U0", "--no-color", "--no-ext-diff", "--no-renames", base, trial],
                         capture_output=True, check=True).stdout.decode("utf-8", errors="replace")
    comp = [(n, re.compile(p)) for n, p in PATTERNS]
    hits, path, ln = [], None, 0
    for line in raw.split("\n"):
        if line.startswith("+++ "):
            path = line[6:] if line.startswith("+++ b/") else None
            continue
        if line.startswith("@@"):
            m = re.match(r"@@ -\S+ \+(\d+)", line)
            ln = int(m.group(1)) if m else 0
            continue
        if line.startswith("+") and path is not None:
            text = line[1:]
            for name, rx in comp:
                for k, m in enumerate(rx.finditer(text)):
                    if k >= 20:
                        break
                    ctx = text[max(0, m.start() - 120): m.end() + 120]
                    hits.append({"path": path, "line": ln, "category": category(path), "pattern": name,
                                 "likely_negated": bool(NEG.search(ctx)), "context": ctx})
            ln += 1
    counts = collections.Counter((h["category"], h["pattern"]) for h in hits)
    code = [h for h in hits if h["category"] in ("src", "tools", "ci")]
    code_unneg = [h for h in code if not h["likely_negated"]]
    by_cat = collections.Counter(h["category"] for h in hits)
    with open(prefix + ".tsv", "w") as f:
        f.write("scope\tcategory\tpattern\tlikely_negated\tpath\tline\tcontext\n")
        for h in hits:
            f.write("\t".join([scope, h["category"], h["pattern"], str(h["likely_negated"]), h["path"], str(h["line"]),
                               h["context"].replace("\t", " ").replace("\r", " ")]) + "\n")
    status = "NO_HITS" if not hits else "HITS_LISTED_FOR_REVIEW"
    emit(prefix + ".json", {"status": status, "scope": scope, "base": base, "trial": trial,
                            "counts": {"total": len(hits), "by_category": dict(by_cat),
                                       "code_ci_hits": len(code), "code_ci_hits_not_negated": len(code_unneg)},
                            "by_category_pattern": {f"{c}|{p}": n for (c, p), n in sorted(counts.items())},
                            "code_ci_hits": code, "hits_tsv": prefix + ".tsv"})
    return 0

def glob_re(p):
    out, i = "", 0
    while i < len(p):
        if p.startswith("**", i):
            out += ".*"
            i += 2
        elif p[i] == "*":
            out += "[^/]*"
            i += 1
        elif p[i] == "?":
            out += "[^/]"
            i += 1
        else:
            out += re.escape(p[i])
            i += 1
    return re.compile("^" + out + "$")


def filter_match(patterns, files):
    rules = [(p.startswith("!"), glob_re(p[1:] if p.startswith("!") else p)) for p in patterns]
    hit = []
    for f in files:
        ok = False
        for neg, rx in rules:
            if rx.match(f):
                ok = not neg
        if ok:
            hit.append(f)
    return hit

                m = filter_match(pr["paths"], files)

                      "macro": got.get("macro"), "values": got, "mismatches": diff})
    emit(out, {"status": overall(items), "items": items})

# ---------------------------------------------------------------- step runner
selected() {
  local n=$1
  if [ -n "$ONLY" ] && ! [[ ",$ONLY," == *",$n,"* ]]; then return 1; fi
  if [ -n "$SKIP" ] && [[ ",$SKIP," == *",$n,"* ]]; then return 1; fi
  return 0
}

record() { # name status rc seconds log note
  STATUS[$1]=$2
  printf '%s\t%s\t%s\t%s\t%s\t%s\n' "$1" "$2" "$3" "$4" "$5" "$6" >> "$STEPS"
}

# run_step NAME FATAL DEPS(comma list or -) FUNC [ARGS...]
run_step() {
  local name=$1 fatal=$2 deps=$3; shift 3
  local always=0
  case "$name" in fetch|verify_inputs|verify_baselines|env_setup|build_trial) always=1 ;; esac
  if [ "$always" = 0 ] && ! selected "$name"; then record "$name" NOT_RUN - 0 - "not selected (--only/--skip)"; return 0; fi
  if [ "$FATAL" = 1 ]; then record "$name" NOT_RUN - 0 - "earlier fatal step failed"; return 0; fi
  local d
  for d in ${deps//,/ }; do
    [ "$d" = "-" ] && continue
    if [ "${STATUS[$d]:-NOT_RUN}" != PASS ]; then record "$name" NOT_RUN - 0 - "dependency $d is ${STATUS[$d]:-NOT_RUN}"; return 0; fi
  done
  STEPNO=$((STEPNO + 1))
  local logf; logf="$OUT/logs/$(printf '%02d' "$STEPNO")_${name}.log"
  log "BEGIN $name"
  local t0 t1 rc
  t0=$(date +%s.%N)
  set +e
  ( set -euo pipefail; "$@" ) > "$logf" 2>&1
  rc=$?
  set -e
  t1=$(date +%s.%N)
  local secs; secs=$("$SYSPY" -c "print(round($t1-$t0,1))")
  local status=FAIL
  if [ "$rc" -eq 0 ]; then
    status=PASS
    local rj="$OUT/results/${name}.json"
    if [ -f "$rj" ]; then
      local s; s=$("$SYSPY" -c 'import json,sys; print(json.load(open(sys.argv[1])).get("status","PASS"))' "$rj")
      # a result file may only refine a PASS (e.g. HITS_LISTED_FOR_REVIEW); it can never turn rc!=0 into PASS
      [ -n "$s" ] && status=$s
    fi
  fi
  record "$name" "$status" "$rc" "$secs" "$logf" ""
  log "END   $name rc=$rc status=$status ${secs}s log=$logf"
  if [ "$rc" -ne 0 ]; then
    log "----- last 25 lines of $logf (display only; rc already recorded) -----"
    tail -n 25 "$logf" | sed 's/^/    | /' >&2 || true
    if [ "$fatal" = 1 ]; then FATAL=1; log "FATAL: $name failed; remaining steps will be NOT_RUN"; fi
  fi
  return 0
}

  local s38 s31 s35; s38=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 1p); s31=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 2p); s35=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 3p)
  if [[ "$LABEL" != DRY_RUN* ]] && [ "$s31" = "$PRE_V2_PR31" ]; then
    echo "REFUSED: label $LABEL is not a dry run but #31 is still at the pre-v2 head $PRE_V2_PR31"; return 1
  fi
  git -C "$REPO" merge-base --is-ancestor "$s31" "$s35" || { echo "FAIL: #31 $s31 is not an ancestor of #35 $s35"; return 1; }
  echo "OK: #31 is an ancestor of #35"
  for c in "$s38" "$s31" "$s35"; do
    git -C "$REPO" merge-base --is-ancestor "$CANON" "$c" || { echo "FAIL: canonical $CANON not ancestor of $c"; return 1; }
    git -C "$REPO" merge-base --is-ancestor "$OWNER_TC" "$c" || { echo "FAIL: Track C owner $OWNER_TC not ancestor of $c"; return 1; }
  done
  echo "OK: canonical b8e39a2 and Track C b9e01a9 are ancestors of all three heads"
  echo "merge-base(#38,#31)=$(git -C "$REPO" merge-base "$s38" "$s31")"
  echo "merge-base(#38,#35)=$(git -C "$REPO" merge-base "$s38" "$s35")"
  echo "#35-own commits (#31..#35): $(git -C "$REPO" rev-list --count "$s31..$s35")"
  echo "#31 commits beyond merge-base with #38: $(git -C "$REPO" rev-list --count "$(git -C "$REPO" merge-base "$s38" "$s31")..$s31")"
}

step_verify_baselines() {
  local ok=0 got
  : > "$OUT/results/baselines.tsv"
  for spec in "canonical|$BR_CANON|$CANON" "trackc_owner|$BR_OWNER_TC|$OWNER_TC" "pr30_base|$BR30|$PR30"; do
    IFS='|' read -r what br want <<< "$spec"
    got=$(remote_head "$br")
    printf '%s\t%s\t%s\t%s\n' "$what" "$br" "$want" "$got" | tee -a "$OUT/results/baselines.tsv"
    [ "$got" = "$want" ] || { echo "BASELINE MOVED: $what $br expected $want got $got"; ok=1; }
  done
  git -C "$REPO" cat-file -e "$MACRO_PIN^{commit}" && echo "macro pin $MACRO_PIN present"
  return $ok
}

step_build_trial() {
  local s38 s31 s35
  s38=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 1p); s31=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 2p); s35=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 3p)
  git -C "$REPO" worktree add --detach "$WT" "$s38"
  cd "$WT"
  export GIT_AUTHOR_DATE="$MERGE_DATE" GIT_COMMITTER_DATE="$MERGE_DATE"
  local G=(git -c gc.auto=0 -c maintenance.auto=false -c commit.gpgsign=false -c user.name="CDR-011 trial harness" -c user.email="trial-harness@invalid")
  local pr sha br out rc
  : > "$OUT/results/merges.tsv"
  for spec in "31|$s31|$BR31" "35|$s35|$BR35"; do
    IFS='|' read -r pr sha br <<< "$spec"
    set +e
    out=$("${G[@]}" merge --no-ff --no-edit -m "[$LABEL] Merge PR #$pr $br @ $sha into trial" "$sha" 2>&1)
    rc=$?
    set -e
    printf '%s\n' "$out"
    printf '%s\n' "$out" | grep -E '^Auto-merging ' > "$OUT/results/automerged_pr$pr.txt" || true
    printf '%s\t%s\t%s\t%s\n' "$pr" "$sha" "$rc" "$(git rev-parse HEAD)" | tee -a "$OUT/results/merges.tsv"
    if [ "$rc" -ne 0 ]; then
      git diff --name-only --diff-filter=U | tee "$OUT/results/conflicts_pr$pr.txt"
      "${G[@]}" merge --abort || true
      echo "MERGE CONFLICT merging PR #$pr"; return 1
    fi
  done
  echo "trial commit $(git rev-parse HEAD) tree $(git rev-parse 'HEAD^{tree}')"
}

step_merge_conflicts() {
  cat "$OUT/results/merges.tsv"
  local n; n=$(find "$OUT/results" -maxdepth 1 -name 'conflicts_pr*.txt' -exec cat {} + | wc -l)
  echo "conflicted paths: $n"; echo "auto-merged (ort) paths:"
  find "$OUT/results" -maxdepth 1 -name 'automerged_pr*.txt' -exec cat {} +
  [ "$n" -eq 0 ]
}

step_tree_identity() {
  local s38 s31 s35 m1 t direct
  s38=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 1p); s31=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 2p); s35=$(cut -f3 "$OUT/results/inputs.tsv" | sed -n 3p)
  m1=$(cut -f4 "$OUT/results/merges.tsv" | sed -n 1p)
  cd "$WT"
  t=$(git rev-parse "$TRIAL^{tree}")
  direct=$(git -c gc.auto=0 merge-tree --write-tree "$s38" "$s35" | sed -n 1p)
  echo "trial tree            $t"
  echo "direct #38+#35 tree   $direct"
  for c in "$s38" "$s31" "$s35" "$PR30"; do echo "tree($c) = $(git rev-parse "$c^{tree}")"; done
  local d1 d2
  d1=$(git diff --name-status "$m1" "$TRIAL" | sha256sum | cut -d' ' -f1)
  d2=$(git diff --name-status "$s31" "$s35" | sha256sum | cut -d' ' -f1)
  echo "second-merge delta name-status sha256 $d1 ; #31..#35 delta $d2"
  echo "first-merge delta files: $(git diff --name-only "$s38" "$m1" | wc -l); #31 vs merge-base files: $(git diff --name-only "$(git merge-base "$s38" "$s31")" "$s31" | wc -l)"
  local new=1
  for c in "$s38" "$s31" "$s35"; do if [ "$t" = "$(git rev-parse "$c^{tree}")" ]; then new=0; fi; done
  echo "trial tree is new (not identical to any input tree already CI-verified): $new  -> full suite required: $new"
  [ "$t" = "$direct" ] && [ "$d1" = "$d2" ]
}

wait_http() {
  local url=$1 i
  for i in $(seq 1 50); do
    if "$SYSPY" -c 'import sys,urllib.request; urllib.request.urlopen(sys.argv[1], timeout=2)' "$url" 2>/dev/null; then return 0; fi
    sleep 0.2
  done
  echo "server did not answer at $url"; return 1
}

port_free() {
  "$SYSPY" -c 'import socket,sys; s=socket.socket(); s.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1); s.bind(("127.0.0.1", int(sys.argv[1]))); s.close()' "$1" \
    || { echo "port $1 is in use; pass --port"; return 1; }
}

step_remote_recheck() {
  local moved=0 br want got
  while IFS=$'\t' read -r pr br want _ _; do
    got=$(remote_head "$br"); echo "PR#$pr $br start=$want now=$got"
    [ "$got" = "$want" ] || { echo "HEAD MOVED: $br"; moved=1; }
  done < "$OUT/results/inputs.tsv"
  while IFS=$'\t' read -r what br want _; do
    got=$(remote_head "$br"); echo "$what $br expected=$want now=$got"
    [ "$got" = "$want" ] || { echo "BASELINE MOVED: $br"; moved=1; }
  done < "$OUT/results/baselines.tsv"
  return $moved
}
