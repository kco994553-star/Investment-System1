def up(v, k):
    for _ in range(abs(k)):
        v = math.nextafter(v, math.inf if k > 0 else -math.inf)
    return v


def defect(a, n, j, k):
    x = [a] * n
    x[j] = up(a, k)
    return x

GRADE_KEYS = ("status", "outcome_class", "k", "q", "rem", "mean_hex", "lrv_hex", "se_hex", "T_hex", "r", "deg", "ties",
              "p", "p_hex", "p_fraction", "codes", "t_hex", "indices_sha256", "pre_f3_cdr010")


def alt_agrees(pri, alt):
    if pri["outcome_class"] != alt["outcome_class"] or pri["status"] != alt["status"]:
        return False
    if pri["outcome_class"] == "STAT":
        return (alt["T_hex"] == pri["T_hex"] and alt["r"] == pri["r"] and alt["deg"] == pri["deg"]
                and alt["ties"] == pri["ties"] and alt["p"].hex() == pri["p_hex"] and alt["codes"] == pri["codes"]
                and alt["t_hex"] == pri["t_hex"])
    if pri["outcome_class"] == "NOT_RUN":
        return alt["deg"] == pri["deg"] and alt["T_hex"] == pri["T_hex"] and alt["codes"] == pri["pre_f3_cdr010"]["codes"]
    return True


def exact_lrv_zero(x, L):
    n = len(x)
    F = [Fraction(v) for v in x]
    m = sum(F, Fraction(0)) / n
    c = F + F
    return all(sum(c[s:s + L], Fraction(0)) == L * m for s in range(n))


def short(o):
    keep = ("status", "outcome_class", "T_hex", "r", "deg", "ties", "p_fraction", "codes")
    return {k: o.get(k) for k in keep if k in o}


def evaluate(rec_in, *, comparators=True, base=None):
    x, L, B, seed = rec_in["x"], rec_in["L"], rec_in["B"], rec_in["seed"]
    xf = [float(v) for v in x]
    rec = {k: rec_in[k] for k in ("id", "family") if k in rec_in}
    rec.update(n=len(xf), L=L, B=B, seed=seed, x_hex=[v.hex() for v in xf])
    for k in ("x_dec", "note", "gie008_expected_p_fraction"):
        if k in rec_in:
            rec[k] = rec_in[k]
    pri = P.mb_v2_cdr012(xf, L, B, seed, trace=False)
    alt = Q.mb_v2_cdr012_alt(xf, L, B, seed)
    rec["cdr012"] = {k: pri[k] for k in GRADE_KEYS if k in pri}
    rec["alt_bit_identical"] = alt_agrees(pri, alt)
    var = {}
    for name, kw in (("square_pow", {"square": "pow"}), ("partial_single_fsum", {"partial_add": "fsum"}),
                     ("grouping_samples", {"grouping": "samples"}), ("predicate_v_positive", {"predicate": "v_positive"}),
                     ("f3_off_cdr010_literal", {"f3": "off"}), ("f3_broad_deg_ge_B_minus_1", {"f3": "broad"}),
                     ("f3_any_degenerate", {"f3": "any"})):
        o = P.mb_v2_cdr012(xf, L, B, seed, trace=False, **kw)
        var[name] = dict(short(o), same_as_cdr012=(o["status"] == pri["status"] and o.get("T_hex") == pri.get("T_hex")
                                                   and o.get("codes") == pri.get("codes")
                                                   and o.get("t_hex") == pri.get("t_hex")))
    rec["variants"] = var
    rec["exact_lrv_zero"] = exact_lrv_zero(xf, L)
    if base is not None:  # battery case: comparators already computed in expected_cdr010.json (same code)
        rec["literal_numpy"] = base["literal_numpy"]
        rec["v1_kernel"] = base["v1_kernel"]
        rec["exact_binary"] = base["exact_binary"]
        rec["exact_decimal"] = base["exact_decimal"]
    elif comparators:
        eb = O.exact_binary(xf, L, B, seed)
        rec["exact_binary"] = {k: eb.get(k) for k in ("status", "r", "deg", "ties", "p_fraction", "codes", "T_sign")}
        ln = O.literal_numpy_run(xf, L, B, seed, per_replicate=True)
        rec["literal_numpy"] = {k: ln.get(k) for k in ("status", "r", "deg", "ties", "p", "p_fraction", "S_plus1_hex",
                                                       "T_hex", "codes", "undefined_statistic")}
        v1 = O.v1_kernel(xf, L, B, seed)
        rec["v1_kernel"] = {k: v1.get(k) for k in ("status", "r", "deg", "ties", "p", "T_hex", "codes")}
    return rec


def check_construction(case, rec):
    """Abort if a dedicated fixture does not have the property it was built for."""
    c = rec["cdr012"]
    pre = P.mb_v2_cdr012([float.fromhex(h) for h in rec["x_hex"]], case["L"], case["B"], case["seed"], f3="off")
    assert pre["status"] == "OK", (case["id"], "CDR-010 status must be OK (se > 0)", pre["status"])
    if case["expect"] == "NOT_RUN":
        assert pre["deg"] == case["B"] and c["outcome_class"] == "NOT_RUN", (case["id"], pre["deg"], case["B"])
    else:
        assert c["outcome_class"] == "STAT", case["id"]
        if "want_deg" in case:
            assert c["deg"] == case["want_deg"], (case["id"], c["deg"], case["want_deg"])
        if case.get("want_near_undefined"):
            assert c["deg"] < case["B"] and abs(float.fromhex(c["T_hex"])) > 1e12, (case["id"], c["deg"], c["T_hex"])
        if case.get("want_pow_p_differs"):
            assert rec["variants"]["square_pow"]["p_fraction"] != c["p_fraction"], case["id"]
    if not rec["alt_bit_identical"]:
        raise AssertionError(("primary and alt disagree", case["id"]))


def sha256_file(p):
    return hashlib.sha256(open(p, "rb").read()).hexdigest()


def main():
    t0 = time.time()
    base = json.load(open(BASE_EXPECTED))
    base_sha = sha256_file(BASE_EXPECTED)
    groups, regrade = {}, {"cases": 0, "cdr010_graded": 0, "became_not_run": [], "changed_other": [],
                           "alt_bit_identical": 0, "cdr010_status_counts": Counter()}
    for g, cases in base["cases"].items():
        out = []
        for bc in cases:
            src = {"id": bc["id"], "family": bc["family"], "x": [float.fromhex(h) for h in bc["x_hex"]], "L": bc["L"],
                   "B": bc["B"], "seed": bc["seed"], "x_dec": bc["x_dec"]}
            if "gie008_expected_p_fraction" in bc:
                src["gie008_expected_p_fraction"] = bc["gie008_expected_p_fraction"]
            rec = evaluate(src, base=bc)
            old, new = bc["cdr010"], rec["cdr012"]
            regrade["cases"] += 1
            regrade["cdr010_status_counts"][old["status"]] += 1
            regrade["cdr010_graded"] += old["status"] == "OK"
            regrade["alt_bit_identical"] += rec["alt_bit_identical"]
            if new["outcome_class"] == "NOT_RUN":
                regrade["became_not_run"].append(bc["id"])
            else:
                same = all(old.get(k) == new.get(k) for k in ("status", "T_hex", "r", "deg", "ties", "p_hex",
                                                              "p_fraction", "codes", "t_hex", "indices_sha256",
                                                              "mean_hex", "lrv_hex", "se_hex"))
                if not same:
                    regrade["changed_other"].append(bc["id"])
            out.append(rec)
        groups[g] = out
    regrade["cdr010_status_counts"] = dict(regrade["cdr010_status_counts"])
    dedicated = {"cdr012_f3_all_degenerate": f3_cases(), "cdr012_f3_boundary": boundary_cases(),
                 "cdr012_f1_square": f1_cases(), "cdr012_near_undefined": near_undefined_cases()}
    for g, cases in dedicated.items():
        out = []
        for case in cases:
            rec = evaluate(case)
            rec["expect"] = case["expect"]
            check_construction(case, rec)
            out.append(rec)
        groups[g] = out
    # detector index
    det = {
        "F1_square_pow": [r["id"] for r in groups["edge_square_axis"]] + [r["id"] for r in groups["cdr012_f1_square"]],
